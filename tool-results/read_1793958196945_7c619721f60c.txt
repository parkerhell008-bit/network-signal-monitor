   348→    private fun hasPermission(permission: String): Boolean =
   349→        ContextCompat.checkSelfPermission(this, permission) == PackageManager.PERMISSION_GRANTED
   350→
   351→    private fun hasPhonePermission(): Boolean =
   352→        hasPermission(Manifest.permission.READ_PHONE_STATE)
   353→
   354→    private fun hasLocationPermission(): Boolean =
   355→        hasPermission(Manifest.permission.ACCESS_FINE_LOCATION) ||
   356→            hasPermission(Manifest.permission.ACCESS_COARSE_LOCATION)
   357→
   358→    private fun missingPermissions(): List<String> {
   359→        val wanted = mutableListOf<String>()
   360→        if (!hasPhonePermission()) wanted.add(Manifest.permission.READ_PHONE_STATE)
   361→        if (!hasLocationPermission()) {
   362→            wanted.add(Manifest.permission.ACCESS_FINE_LOCATION)
   363→            wanted.add(Manifest.permission.ACCESS_COARSE_LOCATION)
   364→        }
   365→        return wanted
   366→    }
   367→
   368→    private fun handlePermissionResult(grants: Map<String, Boolean>) {
   369→        val phoneGranted = grants[Manifest.permission.READ_PHONE_STATE] == true
   370→        val locationGranted = grants[Manifest.permission.ACCESS_FINE_LOCATION] == true ||
   371→            grants[Manifest.permission.ACCESS_COARSE_LOCATION] == true
   372→        when {
   373→            phoneGranted -> {
   374→                startAll()
   375→                // اگر موقعیت رد شده باشد، API 30+ لیست سلول‌ها را خالی برمی‌گرداند؛
   376→                // به‌جای سکوت، راهنمای شفاف نمایش می‌دهیم.
   377→                if (!locationGranted) {
   378→                    toast(getString(R.string.location_needed_for_cells))
   379→                }
   380→            }
   381→            locationGranted -> {
   382→                showPermissionGate(getString(R.string.sim_no_permission))
   383→            }
   384→            else -> {
   385→                showPermissionGate(getString(R.string.permission_gate))
   386→                toast(getString(R.string.permission_gate))
   387→            }
   388→        }
   389→    }

   391→    /** نمایش کادر راهنمای مجوزها؛ لمس آن دیالوگ مجوز را دوباره باز می‌کند */
   392→    private fun showPermissionGate(message: String) {
   393→        tvPermissionGate.visibility = View.VISIBLE
   394→        tvPermissionGate.text = message
   395→        tvPermissionGate.setOnClickListener {
   396→            val missing = missingPermissions()
   397→            if (missing.isEmpty()) {
   398→                startAll()
   399→            } else {
   400→                permissionLauncher.launch(missing.toTypedArray())
   401→            }
   402→        }
   403→    }

   407→    private fun startAll() {
   408→        tvPermissionGate.visibility = View.GONE
   409→        tvPermissionGate.setOnClickListener(null)

   416→        startLiveTraffic()
   417→    }

   419→    private fun stopAll() {
   420→        unregisterSignalListener()
   421→        cellPollJob?.cancel()
   422→        cellPollJob = null
   423→        // توقف کارهای در جریان (تست سرعت / GPS+OpenCelliD) هنگام خروج از صفحه؛
   424→        // اگر تستی در جریان بود، رابط کاربری هم به حالت آماده بازگردانده می‌شود
   425→        // تا دکمه «لغو تست» و نوار پیشرفت پس از بازگشت به صفحه گیر نکنند.
   426→        if (speedTestJob?.isActive == true) {
   427→            speedTestJob?.cancel()
   428→            resetSpeedUi()
   429→            tvSpeedStatus.text = getString(R.string.speed_status_idle)
   430→        }
   431→        speedTestJob = null
   432→        if (distanceJob?.isActive == true) {
   433→            distanceJob?.cancel()
   434→            tvTowerDistance.text = getString(R.string.distance_na)
   435→            tvTowerBearing.text = "—"
   436→        }
   437→        distanceJob = null
   438→        // توقف شمارندهٔ ترافیک لحظه‌ای
   439→        if (liveTrafficJob?.isActive == true) {
   440→            liveTrafficJob?.cancel()
   441→            tvLiveDown.text = getString(R.string.speed_na)
   442→            tvLiveUp.text = getString(R.string.speed_na)
   443→        }
   444→        liveTrafficJob = null
   445→    }

   449→    private fun refreshTelephonyInfo() {
   450→        if (!hasPhonePermission()) {
   451→            tvOperatorName.text = "—"
   452→            tvSimOperator.text = "—"
   453→            tvSimState.text = "—"
   454→            tvCountryIso.text = "—"
   455→            tvRoaming.text = "—"
   456→            tvDataState.text = "—"
   457→            tvDataActivity.text = "—"
   458→            tvPhoneCount.text = "—"
   459→            return
   460→        }
   461→        try {
   462→            val operatorName = telephonyManager.networkOperatorName
   463→                .ifBlank {
   464→                    telephonyManager.simOperatorName
   465→                        .ifBlank { getString(R.string.unknown_operator) }
   466→                }
   467→            tvOperatorName.text = operatorName
   468→            tvSimOperator.text = telephonyManager.simOperatorName
   469→                .ifBlank { getString(R.string.unknown_operator) }
   470→            tvSimState.text = simStateText(telephonyManager.simState)
   471→            tvCountryIso.text = telephonyManager.networkCountryIso
   472→                .uppercase(Locale.US)
   473→                .ifBlank { "—" }
   474→            tvRoaming.text = if (telephonyManager.isNetworkRoaming) {
   475→                getString(R.string.roaming_on)
   476→            } else {
   477→                getString(R.string.roaming_off)
   478→            }
   479→            tvDataState.text = dataStateText(telephonyManager.dataState)
   480→            tvDataActivity.text = dataActivityText(telephonyManager.dataActivity)
   481→            val modems = if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.R) {
   482→                telephonyManager.activeModemCount
   483→            } else {
   484→                @Suppress("DEPRECATION")
   485→                telephonyManager.phoneCount
   486→            }
   487→            tvPhoneCount.text = modems.toPersianDigits()
   488→        } catch (e: SecurityException) {
   489→            tvOperatorName.text = getString(R.string.sim_no_permission)
   490→        }
   491→    }

   582→    private fun registerSignalListener() {
   583→        if (!hasPhonePermission()) return
   584→        // جلوگیری از ثبت تکراری کال‌بک (نشت حافظه) در جریان onResume مجدد
   585→        unregisterSignalListener()
   586→
   587→        // توجه: در SDKهای اخیر، TelephonyCallback.SignalStrengthListener از سطح API عمومی
   588→        // حذف شده است؛ بنابراین در همهٔ نسخه‌ها از PhoneStateListener (منسوخ‌شده اما
   589→        // همچنان موجود در SDK و کارا روی دستگاه‌ها) استفاده می‌شود.
   590→        @Suppress("DEPRECATION")
   591→        val listener = object : PhoneStateListener() {
   592→            override fun onSignalStrengthsChanged(signalStrength: SignalStrength?) {
   593→                // runCatching: برخی سازندگان SignalStrength ناقص می‌دهند؛ هیچ‌گاه کرش نکنیم
   594→                signalStrength?.let { runCatching { onSignalUpdated(it) } }
   595→            }
   596→        }
   597→        phoneListener = listener
   598→        @Suppress("DEPRECATION")
   599→        telephonyManager.listen(listener, PhoneStateListener.LISTEN_SIGNAL_STRENGTHS)
   600→    }
   601→
   602→    private fun unregisterSignalListener() {
   603→        phoneListener?.let {
   604→            @Suppress("DEPRECATION")
   605→            telephonyManager.listen(it, PhoneStateListener.LISTEN_NONE)
   606→        }
   607→        phoneListener = null
   608→    }

   610→    private fun onSignalUpdated(ss: SignalStrength) {
   611→        // در SDKهای اخیر SignalStrength.getDbm()/getAsuLevel() از سطح API عمومی حذف شده‌اند؛
   612→        // dBm و ASU از cellSignalStrengths (API 29+) استخراج می‌شوند و در API 26-28 از فرمول GSM قدیمی.
   613→        // برخی سازندگان cellSignalStrengths خالی برمی‌گردانند؛ در این حالت نیز از فرمول قدیمی استفاده می‌کنیم.
   614→        val dbmRaw = if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.Q) {
   615→            ss.cellSignalStrengths.firstNotNullOfOrNull { css ->
   616→                css.dbm.takeIf { it != Int.MAX_VALUE && it != 0 }
   617→            } ?: legacyGsmDbm(ss)
   618→        } else {
   619→            @Suppress("DEPRECATION")
   620→            legacyGsmDbm(ss)
   621→        }
   622→        val dbm = dbmRaw.takeIf { it != Int.MAX_VALUE && it != 0 }
   623→        lastDbm = dbm
   624→
   625→        dbm?.let { gaugeView.setDbm(it) }
   626→        // SignalStrength.getLevel از API ۲۳ در دسترس است؛ روی اندروید ۵ از dBm تخمین زده می‌شود
   627→        val level = if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.M) {
   628→            ss.level
   629→        } else {
   630→            levelFromDbm(dbm)
   631→        }
   632→        barsView.setLevel(level)
   633→
   634→        tvDbm.text = formatDbmOrNull(dbm)
   635→        tvLevel.text = levelText(level)
   636→
   637→        val asu: Int? = if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.Q) {
   638→            ss.cellSignalStrengths.firstNotNullOfOrNull { it.asuLevel?.takeIf { v -> v in 0..999 } }
   639→        } else null
   640→        tvAsu.text = if (asu != null && asu in 0..999) asu.toPersianDigits() else "—"
   641→
   642→        var rsrp: Int? = null
   643→        var rsrq: Int? = null
   644→        var snr: Int? = null
   645→        var rssi: Int? = null
   646→
   647→        // SignalStrength.cellSignalStrengths از API 29 در دسترس است
   648→        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.Q) {
   649→            for (css in ss.cellSignalStrengths) {
   650→                when (css) {
   651→                    is CellSignalStrengthLte -> {
   652→                        rsrp = validSignal(css.rsrp) ?: rsrp
   653→                        rsrq = validSignal(css.rsrq) ?: rsrq
   654→                        rssi = validSignal(css.rssi) ?: rssi
   655→                        snr = validSignal(css.rssnr) ?: snr
   656→                    }
   657→                    is CellSignalStrengthNr -> {
   658→                        rsrp = validSignal(css.ssRsrp) ?: rsrp
   659→                        rsrq = validSignal(css.ssRsrq) ?: rsrq
   660→                        snr = validSignal(css.ssSinr) ?: snr
   661→                    }
   662→                }
   663→            }
   664→        }
   665→
   666→        tvRsrp.text = formatDbmOrNull(rsrp)
   667→        tvRsrq.text = rsrq?.let { it.toPersianDigits() + " dB" } ?: "—"
   668→        tvSnr.text = snr?.let { it.toPersianDigits() + " dB" } ?: "—"
   669→        tvRssi.text = formatDbmOrNull(rssi)
   670→
   671→        // نمودار زندهٔ سیگنال + امتیاز کیفی
   672→        lastRsrpVal = rsrp
   673→        signalChart.addSample(dbm)
   674→        updateQualityScore(dbm)
   675→    }

   677→    /** امتیاز کیفی ۰ تا ۱۰۰ از dBm/RSRP + برچسب فارسی و رنگ */
   678→    private fun updateQualityScore(dbm: Int?) {
   679→        val gen = lastServingGeneration
   680→        val raw: Double = when {
   681→            lastRsrpVal != null && (gen == "4G" || gen == "5G") -> (lastRsrpVal!! + 120) * 100.0 / 50.0
   682→            dbm != null -> (dbm + 113) * 100.0 / 60.0
   683→            else -> -1.0
   684→        }
   685→        if (raw < 0) {
   686→            tvSignalScore.text = "—"
   687→            return
   688→        }
   689→        val score = raw.toInt().coerceIn(0, 100)
   690→        val label = when {
   691→            score >= 80 -> "عالی"
   692→            score >= 60 -> "خوب"
   693→            score >= 40 -> "متوسط"
   694→            score >= 20 -> "ضعیف"
   695→            else -> "خیلی ضعیف"
   696→        }
   697→        tvSignalScore.text =
   698→            getString(R.string.score_fmt, score.toString().toPersianDigits()) + " — " + label
   699→        tvSignalScore.setTextColor(
   700→            ContextCompat.getColor(
   701→                this,
   702→                when {
   703→                    score >= 60 -> R.color.sig_good
   704→                    score >= 40 -> R.color.sig_mid
   705→                    else -> R.color.sig_bad
   706→                }
   707→            )
   708→        )
   709→    }

   714→    /** تخمین سطح ۰ تا ۴ از dBm برای اندروید ۵ و ۶ که getLevel ندارند */
   715→    private fun levelFromDbm(dbm: Int?): Int = when {
   716→        dbm == null -> -1

   726→     * فرمول استاندارد GSM: dBm = -113 + 2 × ASU
   727→     * توجه: SignalStrength.isGsm فقط از API ۲۶ هست؛ به‌جای آن بازهٔ ASU بررسی می‌شود
   728→     * (روی دستگاه‌های CDMA مقدار ۹۹ برمی‌گردد که نامعتبر است و حذف می‌شود).

   754→    private fun refreshCellInfo() {
   755→        if (!hasPhonePermission()) return
   756→
   757→        val rows = mutableListOf<CellRow>()
   758→        var servingNumeric: NumericServing? = null
   759→
   760→        val all: List<CellInfo>? = try {
   761→            @Suppress("DEPRECATION")
   762→            telephonyManager.allCellInfo
   763→        } catch (e: SecurityException) {
   764→            null
   765→        } catch (e: Exception) {
   766→            null
   767→        }
   768→
   769→        if (!all.isNullOrEmpty()) {
   770→            for (info in all) {
   771→                try {
   772→                    parseCellInfo(info)?.let { rows.add(it) }
   773→                    if (info.isRegistered) {
   774→                        servingNumeric = extractNumeric(info) ?: servingNumeric
   775→                    }
   776→                } catch (t: Throwable) {
   777→                    // بعضی سازندگان CellInfo ناقص برمی‌گردانند؛ این سلول نادیده گرفته می‌شود
   778→                }
   779→            }
   780→        }
   781→
   782→        if (rows.isEmpty()) {
   783→            // جایگزین GsmCellLocation (که از SDK حذف شده): از آخرین قدرت سیگنالِ شنیده‌شده
   784→            // یک ردیف عمومی می‌سازیم تا لیست هرگز خالی نماند؛ شناسهٔ دقیق دکل نیازمند
   785→            // getAllCellInfo کارا است که برخی سازندگان فقط با GPS روشن آن را می‌دهند.
   786→            lastDbm?.let {
   787→                rows.add(
   788→                    CellRow(
   789→                        rat = networkTypeInfo(currentNetworkType(telephonyManager)).first,
   790→                        generation = "",
   791→                        pciOrPsc = "—",
   792→                        cidOrNci = "—",
   793→                        tacOrLac = "—",
   794→                        earfcnOrArfcn = "—",
   795→                        dbm = it,
   796→                        registered = true
   797→                    )
   798→                )
   799→            }
   800→        }
   801→        // در حالت جایگزین، نسل شبکه را از نوع فعلی شبکه پر می‌کنیم تا نشان خالی نماند
   802→        if (rows.size == 1 && rows[0].generation.isBlank()) {
   803→            rows[0] = rows[0].copy(generation = networkTypeInfo(currentNetworkType(telephonyManager)).second)
   804→        }
   805→
   806→        // فاصله محاسبه‌شده (در صورت وجود) به ردیف سروینگ اعمال می‌شود
   807→        val dist = lastDistanceText
   808→        val finalRows = if (dist == null) {
   809→            rows
   810→        } else {
   811→            rows.map { if (it.registered) it.copy(distanceText = dist) else it }
   812→        }
   813→
   814→        cellAdapter.submit(finalRows)
   815→        tvNoCells.visibility = if (finalRows.isEmpty()) View.VISIBLE else View.GONE
   816→
   817→        val serving = finalRows.firstOrNull { it.registered } ?: finalRows.firstOrNull()
   818→        if (serving != null) {
   819→            updateServingTowerCard(serving)
   820→        }
   821→
   822→        numericServing = servingNumeric ?: numericServing
   823→        updateMccMncRow()
   824→
   825→        // محاسبهٔ خودکار فاصله هنگام تغییر دکل متصل (بدون نیاز به لمس دکمه)
   826→        maybeAutoDistance()
   827→    }

  1287→    /**
  1288→     * تخمین فاصله از قدرت سیگنال با مدل تضعیف لگاریتمی (مسیر آزاد):
  1289→     * FSPL(dB) = ۲۰log₁₀(d_km) + ۲۰log₁₀(f_MHz) + ۳۲٫۴۴
  1290→     * ثابت‌های کالیبراسیون شهری (توان مؤثر پرتاب + اتلاف نفوذ) طوری انتخاب شده‌اند
  1291→     * که برای شبکه‌های ایران مقادیر واقع‌بینانه بدهد. خروجی «تخمین تقریبی» است.
  1292→     */
  1293→    private fun estimateDistanceFromSignal(): String? {
  1294→        val gen = lastServingGeneration
  1295→        val perRe = gen == "4G" || gen == "5G"   // RSRP به‌ازای هر المان منبع است
  1296→        val dbm = if (perRe) (lastRsrpVal ?: lastDbm ?: return null) else (lastDbm ?: return null)
  1297→        if (dbm >= 0 || dbm < -140) return null
  1298→
  1299→        val freqMHz = when (gen) {
  1300→            "5G" -> 3500.0
  1301→            "4G" -> 1800.0
  1302→            "3G" -> 2100.0
  1303→            else -> 900.0
  1304→        }
  1305→        // توان مؤثر (EIRP) در گیرنده: LTE/NR هر المان منبع ≈ ۲۷dBm؛ GSM/UMTS کل کانال ≈ ۵۸dBm
  1306→        val eirp = if (perRe) 27.0 else 58.0
  1307→        val fspl = eirp - dbm - 25.0   // ۲۵dB اتلاف شهری/نفوذ ساختمان
  1308→        if (fspl <= 10.0) return formatDistance(40.0)
  1309→
  1310→        val km = Math.pow(10.0, (fspl - 20.0 * Math.log10(freqMHz) - 32.44) / 20.0)
  1311→        val meters = (km * 1000.0).coerceIn(30.0, 50_000.0)
  1312→        return formatDistance(meters)
  1313→    }

  1361→        connection.connectTimeout = 10_000
  1362→        connection.readTimeout = 10_000
  1363→        try {
  1364→            val code = connection.responseCode
  1365→            val stream = if (code in 200..299) connection.inputStream else connection.errorStream
  1366→            val body = stream?.bufferedReader()?.use { it.readText() } ?: ""
  1367→            if (code !in 200..299) {
  1368→                // اگر سرور پیام خطای خوانا فرستاده، همان را نشان می‌دهیم (مثلاً کلید نامعتبر)
  1369→                val serverMessage = runCatching {
  1370→                    val obj = JSONObject(body)
  1371→                    listOf("message", "error", "description")
  1372→                        .firstOrNull { obj.optString(it).isNotEmpty() }
  1373→                        ?.let { k -> obj.optString(k) }
  1374→                }.getOrNull()

  1376→            }
  1377→            val obj = JSONObject(body)
  1378→            if (obj.has("lat") && obj.has("lon")) {
  1379→                return obj.getDouble("lat") to obj.getDouble("lon")
  1380→            }
  1381→            // کلیدهای خطای احتمالی OpenCelliD: message / error / description
  1382→            val message = listOf("message", "error", "description")
  1383→                .firstOrNull { obj.optString(it, "").isNotEmpty() }
  1384→                ?.let { k -> obj.optString(k) }
  1385→                ?: ""
  1386→            throw IOException(
  1387→                if (message.isNotEmpty()) message else "پاسخ نامعتبر از OpenCelliD"
  1388→            )
  1389→        } finally {
  1390→            connection.disconnect()
  1391→        }
  1392→    }

  1394→    /** فرمول هاورساین: فاصلهٔ کروی بین دو نقطه برحسب متر */
  1395→    private fun haversineMeters(lat1: Double, lon1: Double, lat2: Double, lon2: Double): Double {
  1396→        val earthRadius = 6_371_000.0
  1397→        val dLat = Math.toRadians(lat2 - lat1)
  1398→        val dLon = Math.toRadians(lon2 - lon1)
  1399→        val a = sin(dLat / 2) * sin(dLat / 2) +
  1400→            cos(Math.toRadians(lat1)) * cos(Math.toRadians(lat2)) *
  1401→            sin(dLon / 2) * sin(dLon / 2)
  1402→        val c = 2 * atan2(sqrt(a), sqrt(1 - a))
  1403→        return earthRadius * c
  1404→    }

  1407→    private fun bearingDegrees(lat1: Double, lon1: Double, lat2: Double, lon2: Double): Double {
  1408→        val dLon = Math.toRadians(lon2 - lon1)
  1409→        val lat1Rad = Math.toRadians(lat1)
  1410→        val lat2Rad = Math.toRadians(lat2)
  1411→        val y = sin(dLon) * cos(lat2Rad)
  1412→        val x = cos(lat1Rad) * sin(lat2Rad) - sin(lat1Rad) * cos(lat2Rad) * cos(dLon)
  1413→        return (Math.toDegrees(atan2(y, x)) + 360.0) % 360.0
  1414→    }

  1695→    /** گرفتن روز ژولینی از تاریخ میلادی (الگوریتم jalaali) */
  1696→    private fun gregorianToJdn(gy: Int, gm: Int, gd: Int): Int {
  1697→        fun div(a: Int, b: Int) = a / b
  1698→        var d = div((gy + div(gm - 8, 6) + 100100) * 1461, 4) +
  1699→            div(153 * ((gm + 9) % 12) + 2, 5) + gd - 34840408
  1700→        d -= div(div(gy + 100100 + div(gm - 8, 6), 100) * 3, 4) - 752
  1701→        return d
  1702→    }

  1704→    /** گرفتن سال میلادی از روز ژولینی (فقط سال لازم است) */
  1705→    private fun gregorianYearFromJdn(jdn: Int): Int {
  1706→        fun div(a: Int, b: Int) = a / b
  1707→        var j = 4 * jdn + 139361631
  1708→        j += div(div(4 * jdn + 183187720, 146097) * 3, 4) * 4 - 3908
  1709→        val i = div(j % 1461, 4) * 5 + 308
  1710→        val gm = div(i, 153) % 12 + 1
  1711→        return div(j, 1461) - 100100 + div(8 - gm, 6)
  1712→    }
  1713→
  1714→    /** کبیسه‌بینی و نقطهٔ نوین سال جلالی (الگوریتم jalaali): خروجی [کبیسه، سال میلادی، روز اسفند/مارس] */
  1715→    private fun jalCal(jy: Int): IntArray {
  1716→        fun div(a: Int, b: Int) = a / b
  1717→        val breaks = intArrayOf(
  1718→            -61, 9, 38, 199, 426, 686, 756, 818, 1111, 1181, 1210,
  1719→            1635, 2060, 2097, 2192, 2262, 2324, 2394, 2456, 3178
  1720→        )
  1721→        val gy = jy + 621
  1722→        var leapJ = -14
  1723→        var jp = breaks[0]
  1724→        var jump = 0
  1725→        for (i in 1 until breaks.size) {
  1726→            val jm = breaks[i]
  1727→            jump = jm - jp
  1728→            if (jy < jm) break
  1729→            leapJ += div(jump, 33) * 8 + div(jump % 33, 4)
  1730→            jp = jm
  1731→        }
  1732→        var n = jy - jp
  1733→        leapJ += div(n, 33) * 8 + div(n % 33 + 3, 4)
  1734→        if (jump % 33 == 4 && jump - n == 4) leapJ += 1
  1735→        val leapG = div(gy, 4) - div((div(gy, 100) + 1) * 3, 4) - 150
  1736→        val march = 20 + leapJ - leapG
  1737→        if (jump - n < 6) n = n - jump + div(jump + 4, 33) * 33
  1738→        var leap = ((n + 1) % 33 - 1) % 4
  1739→        if (leap == -1) leap = 4
  1740→        return intArrayOf(leap, gy, march)
  1741→    }

  1743→    /** تبدیل تاریخ میلادی به [سال، ماه، روز] هجری شمسی */
  1744→    private fun jalaliFromGregorian(gy: Int, gm: Int, gd: Int): IntArray {
  1745→        fun div(a: Int, b: Int) = a / b
  1746→        val jdn = gregorianToJdn(gy, gm, gd)
  1747→        val gYear = gregorianYearFromJdn(jdn)
  1748→        var jy = gYear - 621
  1749→        val r = jalCal(jy)
  1750→        val jdn1f = gregorianToJdn(r[1], 3, r[2])
  1751→        var k = jdn - jdn1f
  1752→        if (k >= 0) {
  1753→            if (k <= 185) {
  1754→                // ۶ ماه اول سال (۳۱ روزه)
  1755→                return intArrayOf(jy, 1 + k / 31, k % 31 + 1)
  1756→            }
  1757→            k -= 186
  1758→        } else {
  1759→            // سال جلالی قبل
  1760→            jy -= 1
  1761→            k += 179
  1762→            if (r[0] == 1) k += 1
  1763→        }
  1764→        // ۶ ماه دوم سال (۳۰ روزه)
  1765→        return intArrayOf(jy, 7 + k / 30, k % 30 + 1)
  1766→    }
