   356→   356→            hasPermission(Manifest.permission.ACCESS_COARSE_LOCATION)
   357→   357→
   358→   358→    private fun missingPermissions(): List<String> {
   359→   359→        val wanted = mutableListOf<String>()
   360→   360→        if (!hasPhonePermission()) wanted.add(Manifest.permission.READ_PHONE_STATE)
   361→   361→        if (!hasLocationPermission()) {
   362→   362→            wanted.add(Manifest.permission.ACCESS_FINE_LOCATION)
   363→   363→            wanted.add(Manifest.permission.ACCESS_COARSE_LOCATION)
   364→   364→        }
   365→   365→        return wanted
   366→   366→    }
   367→   367→
   368→   368→    private fun handlePermissionResult(grants: Map<String, Boolean>) {
   369→   369→        val phoneGranted = grants[Manifest.permission.READ_PHONE_STATE] == true
   370→   370→        val locationGranted = grants[Manifest.permission.ACCESS_FINE_LOCATION] == true ||
   371→   371→            grants[Manifest.permission.ACCESS_COARSE_LOCATION] == true
   372→   372→        when {
   373→   373→            phoneGranted -> {
   374→   374→                startAll()
   375→   375→                // اگر موقعیت رد شده باشد، API 30+ لیست سلول‌ها را خالی برمی‌گرداند؛
   376→   376→                // به‌جای سکوت، راهنمای شفاف نمایش می‌دهیم.
   377→   377→                if (!locationGranted) {
   378→   378→                    toast(getString(R.string.location_needed_for_cells))
   379→   379→                }
   380→   380→            }
   381→   381→            locationGranted -> {
   382→   382→                showPermissionGate(getString(R.string.sim_no_permission))
   383→   383→            }
   384→   384→            else -> {
   385→   385→                showPermissionGate(getString(R.string.permission_gate))
   386→   386→                toast(getString(R.string.permission_gate))
   387→   387→            }
   388→   388→        }
   389→   389→    }

   419→   419→    private fun stopAll() {
   420→   420→        unregisterSignalListener()

   422→   422→        cellPollJob = null
   423→   423→        // توقف کارهای در جریان (تست سرعت / GPS+OpenCelliD) هنگام خروج از صفحه؛
   424→   424→        // اگر تستی در جریان بود، رابط کاربری هم به حالت آماده بازگردانده می‌شود
   425→   425→        // تا دکمه «لغو تست» و نوار پیشرفت پس از بازگشت به صفحه گیر نکنند.
   426→   426→        if (speedTestJob?.isActive == true) {

   429→   429→            tvSpeedStatus.text = getString(R.string.speed_status_idle)
   430→   430→        }
   431→   431→        speedTestJob = null
   432→   432→        if (distanceJob?.isActive == true) {
   433→   433→            distanceJob?.cancel()
   434→   434→            tvTowerDistance.text = getString(R.string.distance_na)
   435→   435→            tvTowerBearing.text = "—"
   436→   436→        }
   437→   437→        distanceJob = null
   438→   438→        // توقف شمارندهٔ ترافیک لحظه‌ای
   439→   439→        if (liveTrafficJob?.isActive == true) {
   440→   440→            liveTrafficJob?.cancel()
   441→   441→            tvLiveDown.text = getString(R.string.speed_na)
   442→   442→            tvLiveUp.text = getString(R.string.speed_na)
   443→   443→        }
   444→   444→        liveTrafficJob = null
   445→   445→    }

   610→   610→    private fun onSignalUpdated(ss: SignalStrength) {
   611→   611→        // در SDKهای اخیر SignalStrength.getDbm()/getAsuLevel() از سطح API عمومی حذف شده‌اند؛
   612→   612→        // dBm و ASU از cellSignalStrengths (API 29+) استخراج می‌شوند و در API 26-28 از فرمول GSM قدیمی.
   613→   613→        // برخی سازندگان cellSignalStrengths خالی برمی‌گردانند؛ در این حالت نیز از فرمول قدیمی استفاده می‌کنیم.

   615→   615→            ss.cellSignalStrengths.firstNotNullOfOrNull { css ->
   616→   616→                css.dbm.takeIf { it != Int.MAX_VALUE && it != 0 }
   617→   617→            } ?: legacyGsmDbm(ss)
   618→   618→        } else {
   619→   619→            @Suppress("DEPRECATION")
   620→   620→            legacyGsmDbm(ss)
   621→   621→        }
   622→   622→        val dbm = dbmRaw.takeIf { it != Int.MAX_VALUE && it != 0 }

   624→   624→
   625→   625→        dbm?.let { gaugeView.setDbm(it) }
   626→   626→        // SignalStrength.getLevel از API ۲۳ در دسترس است؛ روی اندروید ۵ از dBm تخمین زده می‌شود
   627→   627→        val level = if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.M) {
   628→   628→            ss.level
   629→   629→        } else {
   630→   630→            levelFromDbm(dbm)
   631→   631→        }

   633→   633→
   634→   634→        tvDbm.text = formatDbmOrNull(dbm)
   635→   635→        tvLevel.text = levelText(level)
   636→   636→
   637→   637→        val asu: Int? = if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.Q) {
   638→   638→            ss.cellSignalStrengths.firstNotNullOfOrNull { it.asuLevel?.takeIf { v -> v in 0..999 } }
   639→   639→        } else null
   640→   640→        tvAsu.text = if (asu != null && asu in 0..999) asu.toPersianDigits() else "—"
   641→   641→
   642→   642→        var rsrp: Int? = null
   643→   643→        var rsrq: Int? = null
   644→   644→        var snr: Int? = null
   645→   645→        var rssi: Int? = null
   646→   646→
   647→   647→        // SignalStrength.cellSignalStrengths از API 29 در دسترس است
   648→   648→        if (Build.VERSION.SDK_INT >= Build.VERSION_CODES.Q) {
   649→   649→            for (css in ss.cellSignalStrengths) {
   650→   650→                when (css) {
   651→   651→                    is CellSignalStrengthLte -> {
   652→   652→                        rsrp = validSignal(css.rsrp) ?: rsrp
   653→   653→                        rsrq = validSignal(css.rsrq) ?: rsrq
   654→   654→                        rssi = validSignal(css.rssi) ?: rssi

   658→   658→                        rsrp = validSignal(css.ssRsrp) ?: rsrp
   659→   659→                        rsrq = validSignal(css.ssRsrq) ?: rsrq
   660→   660→                        snr = validSignal(css.ssSinr) ?: snr

   663→   663→            }

   667→   667→        tvRsrq.text = rsrq?.let { it.toPersianDigits() + " dB" } ?: "—"
   668→   668→        tvSnr.text = snr?.let { it.toPersianDigits() + " dB" } ?: "—"
   669→   669→        tvRssi.text = formatDbmOrNull(rssi)
   670→   670→
   671→   671→        // نمودار زندهٔ سیگنال + امتیاز کیفی
   672→   672→        lastRsrpVal = rsrp
   673→   673→        signalChart.addSample(dbm)
   674→   674→        updateQualityScore(dbm)
   675→   675→    }

   696→   696→        }
   697→   697→        tvSignalScore.text =
   698→   698→            getString(R.string.score_fmt, score.toString().toPersianDigits()) + " — " + label
